class Solution {
    List<String> res = new ArrayList<>();
    public List<String> addOperators(String num, int target) {
        dfs(num, target, 0, "", 0, 0);
        return res;
    }
    void dfs(String num, int target, int idx, String path, long eval, long prev) {
        if(idx == num.length()){
            if(eval == target) res.add(path);
            return;
        }
        for(int i = idx; i < num.length(); i++){
            if(i!= idx && num.charAt(idx) == '0') break;
            long curr = Long.parseLong(num.substring(idx, i+1));
            if(idx == 0){
                dfs(num, target, i+1, path + curr, curr, curr);
            } else {
                dfs(num, target, i+1, path + "+" + curr, eval + curr, curr);
                dfs(num, target, i+1, path + "-" + curr, eval - curr, -curr);
                dfs(num, target, i+1, path + "*" + curr, eval - prev + prev * curr, prev * curr);
            }
        }
    }
}
